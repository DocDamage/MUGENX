[State config]
type = VarSet
trigger1 = 1
fvar(26) = 1; 0なら無し　;1なら有り

;MAXゲージ表示位置
[State config]
type = VarSet
trigger1 = 1
fvar(27) = 1.5; ここの数字を変更することによって、MAXゲージの位置を変更できます。

;例として、1にすると、一番下より少し上に表示されます。
;0が画面から見て一番下に表示され、0より大きいほど上に表示されます。
;調整は任意でお願いします。小数点以下が無いと多分動きませんので注意。
;画面からゲージが見えなくなってしまったら、とりあえず0.0を入れてください。
;とりあえず戻るはずです。
